# Month Navigation Sidebar Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add month-grouped content sections and a sticky "Nesta página" sidebar nav to `/blog` and `/library`, delivered in a single commit for easy rollback.

**Architecture:** Both pages get a two-column CSS grid layout (content + sidebar). Content is grouped by month at build/request time using Astro frontmatter logic. A shared scroll-spy pattern (vanilla JS, `getBoundingClientRect`) highlights the active month in the sidebar. The library's existing theme filter is extended to sync sidebar visibility after filtering.

**Tech Stack:** Astro 5, Tailwind CSS, vanilla TypeScript in `<script>` blocks. No new dependencies.

import { getCollection } from 'astro:content';
import BaseLayout from '../../layouts/BaseLayout.astro';

export const prerender = true;

const MONTH_NAMES = ['Janeiro', 'Fevereiro', 'Março', 'Abril', 'Maio', 'Junho',
  'Julho', 'Agosto', 'Setembro', 'Outubro', 'Novembro', 'Dezembro'];

function monthId(date: Date): string {
  const name = MONTH_NAMES[date.getUTCMonth()]
    .normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase();
  return `${date.getUTCFullYear()}-${name}`;
}

function monthLabel(date: Date): string {
  return `${date.getUTCFullYear()} - ${MONTH_NAMES[date.getUTCMonth()]}`;
}

const allPosts = (await getCollection('blog')).sort(
  (a, b) => b.data.date.valueOf() - a.data.date.valueOf()
);

type PostGroup = { id: string; label: string; posts: typeof allPosts };
const groups: PostGroup[] = [];
for (const post of allPosts) {
  const id = monthId(post.data.date);
  const label = monthLabel(post.data.date);
  const g = groups.find(g => g.id === id);
  if (g) g.posts.push(post);
  else groups.push({ id, label, posts: [post] });
}

<script>
  const sections = Array.from(document.querySelectorAll<HTMLElement>('section[data-month]'));
  const navLinks = Array.from(document.querySelectorAll<HTMLAnchorElement>('[data-month-link]'));

  function updateActiveLink() {
    if (!sections.length || !navLinks.length) return;
    let currentId = sections[0].id;
    for (const section of sections) {
      if (section.getBoundingClientRect().top <= 120) {
        currentId = section.id;
      }
    }
    navLinks.forEach(link => {
      const active = link.dataset.monthLink === currentId;
      link.classList.toggle('text-apple-blue', active);
      link.classList.toggle('dark:text-apple-blue-dark', active);
      link.classList.toggle('font-semibold', active);
      link.classList.toggle('text-apple-ink-48', !active);
      link.classList.toggle('dark:text-[#6e6e73]', !active);
    });
  }

  window.addEventListener('scroll', updateActiveLink, { passive: true });
  updateActiveLink();
</script>

import BaseLayout from '../../layouts/BaseLayout.astro';
import ShareButton from '../../components/ShareButton.astro';
import StarRating from '../../components/StarRating.astro';
import libraryData from '../../data/library.json';

const { items } = libraryData;
const themes = [...new Set(items.map((item) => item.theme))].sort();

const MONTH_NAMES = ['Janeiro', 'Fevereiro', 'Março', 'Abril', 'Maio', 'Junho',
  'Julho', 'Agosto', 'Setembro', 'Outubro', 'Novembro', 'Dezembro'];

function monthId(dateStr: string): string {
  const [year, month] = dateStr.split('-').map(Number);
  const name = MONTH_NAMES[month - 1]
    .normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase();
  return `${year}-${name}`;
}

function monthLabel(dateStr: string): string {
  const [year, month] = dateStr.split('-').map(Number);
  return `${year} - ${MONTH_NAMES[month - 1]}`;
}

type Item = typeof items[0];
type ItemGroup = { id: string; label: string; items: Item[] };

const sortedItems = [...items].sort(
  (a, b) => new Date(b.addedAt).valueOf() - new Date(a.addedAt).valueOf()
);

const groups: ItemGroup[] = [];
for (const item of sortedItems) {
  const id = monthId(item.addedAt);
  const label = monthLabel(item.addedAt);
  const g = groups.find(g => g.id === id);
  if (g) g.items.push(item);
  else groups.push({ id, label, items: [item] });
}

const typeLabel: Record<string, string> = {
  article: 'Artigo',
  video: 'Vídeo',
  link: 'Link',
};

<script>
  // --- Filter ---
  const filterBtns = document.querySelectorAll<HTMLButtonElement>('.filter-btn');
  const libraryItems = document.querySelectorAll<HTMLLIElement>('.library-item');

  function syncMonthVisibility() {
    document.querySelectorAll<HTMLElement>('section[data-month]').forEach(section => {
      const hasVisible = Array.from(section.querySelectorAll<HTMLElement>('.library-item'))
        .some(item => item.style.display !== 'none');
      section.style.display = hasVisible ? '' : 'none';
      const navLi = document.querySelector<HTMLElement>(`[data-month-link="${section.id}"]`)
        ?.closest('li') as HTMLElement | null;
      if (navLi) navLi.style.display = hasVisible ? '' : 'none';
    });
    updateActiveLink();
  }

  filterBtns.forEach((btn) => {
    btn.addEventListener('click', () => {
      filterBtns.forEach((b) => {
        b.classList.remove('bg-apple-blue', 'text-white', 'border-apple-blue');
        b.classList.add('border-apple-hairline', 'text-apple-ink');
      });
      btn.classList.add('bg-apple-blue', 'text-white', 'border-apple-blue');
      btn.classList.remove('border-apple-hairline', 'text-apple-ink');

      const theme = btn.dataset.theme;
      libraryItems.forEach((item) => {
        item.style.display =
          theme === 'all' || item.dataset.theme === theme ? '' : 'none';
      });
      syncMonthVisibility();
    });
  });

  // --- Scroll-spy ---
  const sections = Array.from(document.querySelectorAll<HTMLElement>('section[data-month]'));
  const navLinks = Array.from(document.querySelectorAll<HTMLAnchorElement>('[data-month-link]'));

  function updateActiveLink() {
    if (!navLinks.length) return;
    const visibleSections = sections.filter(s => s.style.display !== 'none');
    if (!visibleSections.length) return;
    let currentId = visibleSections[0].id;
    for (const section of visibleSections) {
      if (section.getBoundingClientRect().top <= 120) {
        currentId = section.id;
      }
    }
    navLinks.forEach(link => {
      const active = link.dataset.monthLink === currentId;
      link.classList.toggle('text-apple-blue', active);
      link.classList.toggle('dark:text-apple-blue-dark', active);
      link.classList.toggle('font-semibold', active);
      link.classList.toggle('text-apple-ink-48', !active);
      link.classList.toggle('dark:text-[#6e6e73]', !active);
    });
  }

  window.addEventListener('scroll', updateActiveLink, { passive: true });
  updateActiveLink();
</script>
