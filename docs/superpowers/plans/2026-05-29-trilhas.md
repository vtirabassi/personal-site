> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `/trilhas` section with curated learning tracks, linear module-level progress tracking via localStorage, and the full CCA Foundation (Anthropic) track as the first entry.

**Architecture:** Two Astro content collections (`tracks` for track metadata, `track-modules` for individual modules), static SSG pages with client-side hydration for progress state. Progress stored in localStorage as `{ completedModules: number[], lastUpdated: number }` keyed by track slug. No backend required.

**Tech Stack:** Astro 5 (content collections, SSG), Tailwind CSS (Apple design tokens), TypeScript, Vitest

- [ ] **Step 3: Create the utility**

Create `src/lib/track-progress.ts`:

```ts
export interface TrackProgress {
  completedModules: number[];
  lastUpdated: number;
}

export function getCompletedModules(slug: string): number[] {
  try {
    const raw = localStorage.getItem(`track-progress-${slug}`);
    if (!raw) return [];
    const data = JSON.parse(raw) as TrackProgress;
    return Array.isArray(data.completedModules) ? data.completedModules : [];
  } catch {
    return [];
  }
}

export function markModuleComplete(slug: string, order: number): void {
  const completed = getCompletedModules(slug);
  if (completed.includes(order)) return;
  const data: TrackProgress = {
    completedModules: [...completed, order].sort((a, b) => a - b),
    lastUpdated: Date.now(),
  };
  localStorage.setItem(`track-progress-${slug}`, JSON.stringify(data));
}

export function calcProgressPercent(completedCount: number, totalCount: number): number {
  if (totalCount === 0) return 0;
  return Math.round((completedCount / totalCount) * 100);
}
```

<script>
  import { getCompletedModules, markModuleComplete, calcProgressPercent } from '../../lib/track-progress';

  const page = document.getElementById('track-page')!;
  const slug         = page.dataset.slug!;
  const totalModules = parseInt(page.dataset.total!);

  function updateProgress() {
    const completed = getCompletedModules(slug);
    const pct = calcProgressPercent(completed.length, totalModules);

    const progressText = document.getElementById('modules-progress-text');
    const progressBar  = document.getElementById('track-progress-bar');
    const pctEl        = document.getElementById('track-pct');
    if (progressText) progressText.textContent = `${completed.length} de ${totalModules} módulos`;
    if (progressBar)  progressBar.style.width = `${pct}%`;
    if (pctEl)        pctEl.textContent = `${pct}%`;

    const moduleEls = document.querySelectorAll<HTMLElement>('[data-module]');
    let currentFound = false;

    moduleEls.forEach(el => {
      const order   = parseInt(el.dataset.order!);
      const isDone  = completed.includes(order);
      const isCurrent = !isDone && !currentFound;
      if (isCurrent) currentFound = true;

      const indicator  = document.getElementById(`indicator-${order}`);
      const content    = document.getElementById(`content-${order}`);
      const completeBtn = document.getElementById(`complete-btn-${order}`);

      if (indicator) {
        if (isDone) {
          indicator.className = 'w-4 h-4 rounded-full bg-apple-blue flex-shrink-0';
        } else if (isCurrent) {
          indicator.className = 'w-4 h-4 rounded-full border-2 border-apple-blue bg-white dark:bg-[#1c1c1e] flex-shrink-0';
        } else {
          indicator.className = 'w-4 h-4 rounded-full border border-apple-hairline bg-apple-divider dark:bg-[#3a3a3c] flex-shrink-0';
        }
      }

      // Card border style
      if (isCurrent) {
        el.classList.add('border-apple-blue', 'border-2');
        el.classList.remove('border-apple-hairline', 'dark:border-[#3a3a3c]');
      } else {
        el.classList.remove('border-apple-blue', 'border-2');
        el.classList.add('border-apple-hairline', 'dark:border-[#3a3a3c]');
      }

      // Auto-open current, collapse done
      if (content) {
        if (isCurrent) content.classList.remove('hidden');
        else if (isDone) content.classList.add('hidden');
      }

      // Show "Marcar como concluído" only for current module
      if (completeBtn) {
        if (isCurrent) completeBtn.classList.remove('hidden');
        else           completeBtn.classList.add('hidden');
      }
    });
  }

  // Expose to inline onclick handlers (required because onclick attrs call globals, not module scope)
  (window as any).__completeModule = (order: number) => {
    markModuleComplete(slug, order);
    updateProgress();
  };

  (window as any).__toggleModule = (order: number) => {
    document.getElementById(`content-${order}`)?.classList.toggle('hidden');
  };

  updateProgress();
</script>
